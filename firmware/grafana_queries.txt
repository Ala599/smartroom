=== PANEL 5 : Historique CO2 - 24h (Time Series) ===
from(bucket: "sensors")
  |> range(start: -24h)
  |> filter(fn: (r) => r._measurement == "environment" and r._field == "co2")
  |> aggregateWindow(every: 5m, fn: mean, createEmpty: false)
  |> yield(name: "co2_mean")

=== PANEL 7 : Taux d'occupation par heure ===
from(bucket: "sensors")
  |> range(start: -24h)
  |> filter(fn: (r) => r._measurement == "environment" and r._field == "pres")
  |> aggregateWindow(every: 1h, fn: sum, createEmpty: false)
  |> map(fn: (r) => ({ r with _value: r._value * 100.0 / 120.0 }))
  |> yield(name: "occupation_rate")
